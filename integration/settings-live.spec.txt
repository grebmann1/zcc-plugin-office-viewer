import { cpSync, mkdirSync, readFileSync, writeFileSync } from "node:fs";
import { join, basename, resolve } from "node:path";
import { execFile } from "node:child_process";
import { promisify } from "node:util";
import { test, expect } from "./fixtures/app.js";

test.use({
  launchEnv: { ZCC_FAKE_PROVIDER: "1", ZCC_SKIP_PLUGIN_NPM: "1" },
  initialConfig: { sponsorPromptDismissed: true },
  isolateBundledCatalog: true,
});
test("Office Viewer uses native Settings and keeps document previews working", async ({ app, home }, testInfo) => {
  test.setTimeout(180000);
  const page = app.window;
  await app.electron.evaluate(({ BrowserWindow }) => BrowserWindow.getAllWindows()[0].setContentSize(1350, 1000));
  const cachedToolchain = process.env.OFFICE_VIEWER_TOOLCHAIN_CACHE;
  if (cachedToolchain) cpSync(cachedToolchain, join(home, ".zcc", "plugins", basename(cachedToolchain)), { recursive: true });
  const target = join(home, "office-viewer-source");
  cpSync(process.env.OFFICE_VIEWER_ROOT!, target, {
    recursive: true,
    filter: (p) => !["node_modules", ".git", "coverage"].includes(basename(p)),
  });
  const cli = resolve(process.env.ZCC_E2E_APP_ROOT ?? ".", "packages/cli/dist/bin/zcc");
  const env = {
    ...process.env, HOME: home, ZCC_DATA_DIR: join(home, ".zcc"),
    ZCC_SERVER_URL: new URL(page.url()).origin,
    ZCC_SESSION_ID: undefined, ZCC_SESSION_TOKEN: undefined,
    ZCC_SKIP_PLUGIN_NPM: "1", npm_config_registry: "https://registry.npmjs.org/",
  };
  const run = (args: string[]) => promisify(execFile)(process.execPath, [cli, ...args], {
    cwd: target, env, timeout: 120000, maxBuffer: 2 * 1024 * 1024,
  });
  expect(JSON.parse((await run(["plugin", "install", target])).stdout)).toMatchObject({ id: "office-viewer", status: "running" });
  const navigate = async (path: string) => page.evaluate(path => {
    history.pushState({}, "", path); dispatchEvent(new PopStateEvent("popstate"));
  }, path);
  await navigate("/settings/general");
  await page.getByTestId("settings-plugin-office-viewer").click();
  await expect(page).toHaveURL(/\/settings\/plugins\/office-viewer$/);
  await expect(page.getByRole("heading", { name: "Office Viewer", exact: true })).toHaveCount(1);
  const preferences = page.getByLabel("Office Viewer preferences");
  await expect(preferences).toBeVisible();
  await expect(preferences.getByRole("region", { name: "Supported documents" })).toBeVisible();
  await expect(preferences.getByRole("definition")).toHaveText([".docx", ".xlsx", ".pptx", "Read-only", "8 MiB"]);
  await expect(preferences.locator(".settings-section-body")).toHaveCount(3);
  await expect(preferences.getByText(/No additional configuration is required/)).toBeVisible();
  const guide = preferences.locator("details");
  await guide.locator("summary").click();
  await expect(guide.getByText(/Open a .docx/)).toBeVisible();
  await page.screenshot({ path: testInfo.outputPath("office-viewer-settings.png") });
  await app.electron.evaluate(({ BrowserWindow }) => BrowserWindow.getAllWindows()[0].setContentSize(900, 800));
  expect(await preferences.evaluate(el => el.scrollWidth <= el.clientWidth)).toBe(true);
  await page.screenshot({ path: testInfo.outputPath("office-viewer-settings-narrow.png") });
  await app.electron.evaluate(({ BrowserWindow }) => BrowserWindow.getAllWindows()[0].setContentSize(1350, 1000));
  const source = join(target, "app.tsx");
  writeFileSync(source, readFileSync(source, "utf8").replace("Supported files open automatically.", "Supported files open automatically. Reload verified."));
  await run(["plugin", "dev", "--once"]);
  await expect(preferences.getByText(/Reload verified/)).toBeVisible();
  await expect(page).toHaveURL(/\/settings\/plugins\/office-viewer$/);
  const projectPath = join(home, "office-review");
  mkdirSync(projectPath);
  for (const file of ["report.docx", "forecast.xlsx", "briefing.pptx"])
    writeFileSync(join(projectPath, file), Buffer.from(readFileSync(join(process.env.OFFICE_VIEWER_ROOT!, "integration", file + ".base64"), "utf8"), "base64"));
  const ids = await page.evaluate(async root => {
    const project = await window.cc.projects.add(root);
    if (!project.ok) throw new Error(project.message);
    const response = await fetch("/api/v1/threads", { method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify({ projectId: project.value.id, providerId: "fake", input: "Review the Office documents" }) });
    const body = await response.json();
    if (!response.ok) throw new Error(JSON.stringify(body));
    return { projectId: project.value.id, threadId: (body.thread ?? body.value).id };
  }, projectPath);
  await expect.poll(() => page.evaluate(async id => (await (await fetch(`/api/v1/threads/${id}`)).json()).thread.status, ids.threadId)).toBe("idle");
  await navigate(`/threads/${ids.threadId}`);
  for (const [file, text] of [["report.docx", "Business proposal"], ["forecast.xlsx", "Quarter"], ["briefing.pptx", "Delivery overview"]]) {
    expect(await page.evaluate(async ({ id, file }) => (await fetch(`/api/v1/threads/${id}/open`, { method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify({ file: { source: "workspace", path: file } }) })).status, { id: ids.threadId, file })).toBe(200);
    const panel = page.getByTestId("thread-secondary-panel");
    await expect(panel.getByText(text, { exact: true })).toBeVisible();
    await panel.getByRole("button", { name: "Quote section", exact: true }).click();
    await expect(page.getByTestId("thread-command-input").first()).toContainText(text);
  }
  await page.screenshot({ path: testInfo.outputPath("office-viewer-preview.png") });
  expect(await page.evaluate(async ids => {
    try {
      await window.cc.pluginApps.callRpc("office-viewer", "preview", { path: "/etc/passwd.docx", source: { kind: "workspace", projectId: ids.projectId, threadId: ids.threadId, environmentId: null } });
      return false;
    } catch { return true; }
  }, ids)).toBe(true);
});
